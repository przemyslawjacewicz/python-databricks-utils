def test_collapse_dataframe__StructField():
    assertDataFrameEqual(
        collapse_dataframe(
            spark.createDataFrame(
                [(1,)],
                StructType([StructField("_1", LongType())])
            ),
            "@"
        ),
        spark.createDataFrame([(1,)])
    )
